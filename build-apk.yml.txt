name: Construire l'APK See Cat

# Se lance tout seul à chaque envoi de fichiers, ou à la main (onglet Actions > Run workflow).
on:
  push:
    branches: [main, master]
  workflow_dispatch:

jobs:
  apk:
    runs-on: ubuntu-latest
    timeout-minutes: 40
    steps:
      - name: Récupérer les fichiers
        uses: actions/checkout@v4

      - name: Installer Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      - name: Installer Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Installer le SDK Android
        uses: android-actions/setup-android@v3

      - name: Installer Capacitor
        run: npm ci

      - name: Créer le projet Android à partir du site (dossier www)
        run: npx cap add android

      - name: Icône, écran de démarrage et numéro de version
        run: |
          cp -r android-res/. android/app/src/main/res/
          sed -i "s/versionCode 1/versionCode ${{ github.run_number }}/" android/app/build.gradle
          sed -i "s/versionName \"1.0\"/versionName \"1.0.${{ github.run_number }}\"/" android/app/build.gradle

      - name: Copier la dernière version du site dans l'app
        run: npx cap sync android

      - name: Compiler l'APK
        working-directory: android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon

      - name: Préparer le fichier à télécharger
        run: cp android/app/build/outputs/apk/debug/app-debug.apk SeeCat.apk

      - name: Mettre l'APK à disposition
        uses: actions/upload-artifact@v4
        with:
          name: SeeCat-APK
          path: SeeCat.apk
          if-no-files-found: error
